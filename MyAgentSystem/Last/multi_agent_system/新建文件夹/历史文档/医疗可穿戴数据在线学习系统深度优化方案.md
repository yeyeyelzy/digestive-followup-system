```Python
import os
from datetime import datetime, timedelta

def generate_patient_dir(patient_id, root_path="./Fitabase Data 4.12.16-5.12.16"):
    """
    生成单患者标准化目录结构
    :param patient_id: 患者ID，如P_001_Wang
    :param root_path: 数据根目录
    """
    # 根目录定义
    patient_root = os.path.join(root_path, patient_id)
    # 一级文件夹
    first_level_dirs = ["raw_data", "patient_end", "family_end", "doctor_end", "system_check", "model_cache"]
    # 端内二级文件夹（日/周/月）
    time_level_dirs = ["daily", "weekly", "monthly"]
    
    # 循环创建目录
    for first_dir in first_level_dirs:
        dir_path = os.path.join(patient_root, first_dir)
        # 4大端需要创建日/周/月二级目录
        if first_dir in ["patient_end", "family_end", "doctor_end", "system_check"]:
            for time_dir in time_level_dirs:
                time_dir_path = os.path.join(dir_path, time_dir)
                os.makedirs(time_dir_path, exist_ok=True)
        else:
            os.makedirs(dir_path, exist_ok=True)
    
    print(f"患者{patient_id}标准化目录生成完成，路径：{patient_root}")

# 批量生成示例
if __name__ == "__main__":
    patient_list = ["P_001_Wang", "P_002"]
    for patient in patient_list:
        generate_patient_dir(patient)
```

### 3.2 核心痛点解决：心率预警误报的归因分析（医生端核心优化）

针对你提出的「医生看到大量红预警，分不清是运动还是病理原因，容易漏诊」的问题，通过**多指标联动归因算法**，实现心率预警的自动分类，并用饼图可视化占比，让医生一眼抓住核心病理性异常。

#### 3.2.1 预警归因算法逻辑

1. 对每一个心率超阈值的时间点，同步匹配同时间窗口的METs、步数、活动强度数据

2. 分类规则（可基于患者基线在线更新）：

    - 生理性预警：同时间窗口METs≥3，或步数>50步/分钟，或活动强度为中/高强度 → 运动/活动导致的心率升高

    - 病理性预警：同时间窗口METs<1.5，且步数<10步/分钟，且活动强度为久坐/轻强度 → 静息状态下的异常心率升高

    - 待排查预警：介于两者之间的边界值，标记为需人工复核

3. 每日/每周汇总所有预警，生成分类统计饼图，放在医生端报告的最开头，实现「先看汇总，再看细节」的详略得当设计。

#### 3.2.2 归因分析Python代码实现

```Python

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
plt.rcParams["font.sans-serif"] = ["SimHei"]
plt.rcParams["axes.unicode_minus"] = False

def heartrate_alarm_attribution(heartrate_df, mets_df, steps_df, intensity_df, rest_hr_threshold=100):
    """
    心率预警归因分析，区分生理性/病理性/待排查预警
    :param heartrate_df: 秒级心率数据，需包含Time、Value列
    :param mets_df: 分钟级METs数据，需包含Time、Value列
    :param steps_df: 分钟级步数数据，需包含Time、Value列
    :param intensity_df: 分钟级强度数据，需包含Time、Value列
    :param rest_hr_threshold: 静息心率预警阈值，可个性化调整
    :return: 预警分类结果、统计占比
    """
    # 时间格式统一，对齐到分钟级
    heartrate_df["Time"] = pd.to_datetime(heartrate_df["Time"]).dt.floor("min")
    mets_df["Time"] = pd.to_datetime(mets_df["Time"]).dt.floor("min")
    steps_df["Time"] = pd.to_datetime(steps_df["Time"]).dt.floor("min")
    intensity_df["Time"] = pd.to_datetime(intensity_df["Time"]).dt.floor("min")
    
    # 合并全维度分钟级数据
    merged_df = heartrate_df.groupby("Time")["Value"].mean().reset_index(name="HeartRate")
    merged_df = merged_df.merge(mets_df, on="Time", how="left", suffixes=("", "_METs"))
    merged_df = merged_df.merge(steps_df, on="Time", how="left", suffixes=("", "_Steps"))
    merged_df = merged_df.merge(intensity_df, on="Time", how="left", suffixes=("", "_Intensity"))
    merged_df = merged_df.fillna(0)
    
    # 筛选心率超阈值的预警数据
    alarm_df = merged_df[merged_df["HeartRate"] >= rest_hr_threshold].copy()
    
    # 归因分类
    def attribution_label(row):
        if row["Value_METs"] >= 3 or row["Value_Steps"] >= 50 or row["Value_Intensity"] >= 2:
            return "生理性预警（运动/活动）"
        elif row["Value_METs"] < 1.5 and row["Value_Steps"] < 10 and row["Value_Intensity"] <= 1:
            return "病理性预警（静息异常）"
        else:
            return "待排查预警（边界值）"
    
    alarm_df["attribution"] = alarm_df.apply(attribution_label, axis=1)
    
    # 统计占比
    alarm_stats = alarm_df["attribution"].value_counts()
    total_alarm = len(alarm_df)
    alarm_ratio = (alarm_stats / total_alarm * 100).round(2)
    
    return alarm_df, alarm_stats, alarm_ratio

# 可视化：医生端预警分类饼图
def plot_alarm_pie(alarm_ratio, save_path, report_date):
```

### 3.3 核心数学模型升级：多指标联动异常检测

原有方案仅用单指标固定阈值，优化后采用**基于孤立森林的增量在线异常检测模型**，融合心率、HRV、METs、步数、睡眠等多维度特征，实现更精准的异常识别，而非单指标的简单超限。

```Python

from sklearn.ensemble import IsolationForest
import joblib
import os

def incremental_anomaly_detection(feature_df, patient_id, model_cache_path, window_size=7, contamination=0.01):
    """
    增量在线异常检测，每日更新模型，符合在线学习要求
    :param feature_df: 当日多维度特征数据，每行1个时间窗口的特征
    :param patient_id: 患者ID
    :param model_cache_path: 模型缓存路径
    :param window_size: 滑动窗口大小（天），仅用近7天数据增量训练，不回溯全量
    :param contamination: 异常比例，可个性化调整
    :return: 异常检测结果、更新后的模型
    """
    model_file = os.path.join(model_cache_path, "anomaly_detection_model.pkl")
    baseline_file = os.path.join(model_cache_path, "baseline_history.csv")
    
    # 1. 加载已有模型/初始化新模型
    if os.path.exists(model_file):
        model = joblib.load(model_file)
        # 在线学习：用当日数据增量更新（孤立森林支持部分拟合，也可采用滑动窗口重新训练，更适合医疗场景）
        # 医疗场景优先用滑动窗口，避免异常数据污染基线，仅用近7天正常数据训练
        baseline_history = pd.read_csv(baseline_file)
        train_data = pd.concat([baseline_history.tail(window_size*1440), feature_df], axis=0) # 1440=分钟/天
    else:
        model = IsolationForest(n_estimators=100, contamination=contamination, random_state=42, n_jobs=-1)
        train_data = feature_df
        baseline_history = pd.DataFrame()
    
    # 2. 模型训练与异常预测
    model.fit(train_data)
    feature_df["anomaly_label"] = model.predict(feature_df)
    feature_df["anomaly_score"] = model.decision_function(feature_df)
    # -1=异常，1=正常
    feature_df["anomaly_label"] = feature_df["anomaly_label"].map({1: "正常", -1: "异常"})
    
    # 3. 保存更新后的模型与基线数据（在线学习痕迹留存）
    joblib.dump(model, model_file)
    updated_baseline = pd.concat([baseline_history, feature_df[feature_df["anomaly_label"]=="正常"]], axis=0)
    updated_baseline.to_csv(baseline_file, index=False)
    
    return feature_df, model
```

### 5.1 在线学习核心技术框架

采用**医疗场景适配的指数移动平均（EMA）增量更新算法**，优势：

1. 无需全量数据回溯，仅用新数据增量更新基线，符合在线学习的「流式数据处理」要求

2. 给近期数据更高权重，适配人体生理状态的渐进式变化，避免历史数据过度影响当前基线

3. 计算效率极高，适合秒级/分钟级流式数据实时处理

4. 全流程更新痕迹可留存，完全满足你「看到基线怎么更新」的需求

```Python

import pandas as pd
import numpy as np

class OnlinePersonalizedBaseline:
    """
    在线个性化基线更新类，核心实现流式数据的增量基线更新
    """
    def __init__(self, patient_id, model_cache_path, alpha=0.3):
        """
        初始化
        :param patient_id: 患者ID
        :param model_cache_path: 模型缓存路径
        :param alpha: EMA平滑系数，0<alpha<1，越大表示近期数据权重越高，医疗场景推荐0.2-0.4
        """
        self.patient_id = patient_id
        self.model_cache_path = model_cache_path
        self.alpha = alpha
        self.baseline_file = os.path.join(model_cache_path, "baseline_history.csv")
        # 核心指标初始化
        self.indicators = [
            "rest_heart_rate", "hrv_sdnn", "hrv_rmssd", 
            "deep_sleep_ratio", "daily_steps", "moderate_high_activity_min",
            "sedentary_min", "sleep_efficiency"
        ]
        # 加载历史基线
        self.baseline_history = self._load_baseline_history()
    
    def _load_baseline_history(self):
        """加载历史基线数据，无则初始化空DataFrame"""
        if os.path.exists(self.baseline_file):
            return pd.read_csv(self.baseline_file, parse_dates=["Date"])
        else:
            return pd.DataFrame(columns=["Date", "patient_id", "indicator", "baseline_mean", "baseline_std", "baseline_upper", "baseline_lower"])
    
    def _get_initial_baseline(self, indicator_data):
        """初始化基线，用前7天数据计算初始均值和标准差"""
        mean = np.mean(indicator_data)
        std = np.std(indicator_data)
        # 95%置信区间作为基线上下限
        upper = mean + 1.96 * std
        lower = mean - 1.96 * std
        return mean, std, upper, lower
    
    def update_baseline_daily(self, date, daily_indicator_data):
        """
        每日增量更新基线，核心在线学习方法
        :param date: 日期，如2016-04-15
        :param daily_indicator_data: 当日指标数据，dict格式，key为指标名，value为当日数值
        :return: 更新后的基线数据
        """
        updated_rows = []
        date = pd.to_datetime(date)
        
        for indicator in self.indicators:
            # 跳过当日无数据的指标
            if indicator not in daily_indicator_data:
                continue
            current_value = daily_indicator_data[indicator]
            
            # 筛选该指标的历史基线
            indicator_history = self.baseline_history[self.baseline_history["indicator"] == indicator]
            
            # 无历史数据：初始化基线
            if len(indicator_history) == 0:
                mean, std, upper, lower = self._get_initial_baseline([current_value])
            # 有历史数据：EMA增量更新
            else:
                last_mean = indicator_history.iloc[-1]["baseline_mean"]
                last_std = indicator_history.iloc[-1]["baseline_std"]
                # EMA增量更新均值和标准差
                mean = self.alpha * current_value + (1 - self.alpha) * last_mean
                std = np.sqrt(self.alpha * (current_value - mean)**2 + (1 - self.alpha) * last_std**2)
                # 更新95%置信区间
                upper = mean + 1.96 * std
                lower = mean - 1.96 * std
            
            # 保存更新后的基线
            updated_row = {
                "Date": date,
                "patient_id": self.patient_id,
                "indicator": indicator,
                "baseline_mean": mean,
                "baseline_std": std,
                "baseline_upper": upper,
                "baseline_lower": lower,
                "daily_value": current_value
            }
            updated_rows.append(updated_row)
        
        # 合并到历史基线并保存
        updated_df = pd.DataFrame(updated_rows)
        self.baseline_history = pd.concat([self.baseline_history, updated_df], axis=0, ignore_index=True)
        self.baseline_history.to_csv(self.baseline_file, index=False)
        
        print(f"{date} 患者{self.patient_id} 基线在线更新完成，共更新{len(updated_rows)}个指标")
        return updated_df

# 调用示例
```

```Python
    :param content_dict: 报告内容字典，包含标题、总结、指标数据、图片路径
    :param save_path: PDF保存路径
    """
    # 初始化PDF文档
    doc_file = os.path.join(save_path, f"{report_date}_{end_type}.pdf")
    doc = SimpleDocTemplate(doc_file, pagesize=A4, rightMargin=2*cm, leftMargin=2*cm, topMargin=2*cm, bottomMargin=2*cm)
    styles = getSampleStyleSheet()
    story = []
    
    # 自定义样式
    title_style = ParagraphStyle(
        "title", parent=styles["Heading1"], fontSize=18, textColor=colors.HexColor("#2c3e50"), alignment=1, spaceAfter=0.5*cm
    )
    subtitle_style = ParagraphStyle(
        "subtitle", parent=styles["Heading2"], fontSize=14, textColor=colors.HexColor("#34495e"), spaceAfter=0.3*cm
    )
    normal_style = ParagraphStyle(
        "normal", parent=styles["Normal"], fontSize=10, spaceAfter=0.2*cm, leading=14
    )
    highlight_style = ParagraphStyle(
        "highlight", parent=styles["Normal"], fontSize=12, textColor=colors.HexColor("#e74c3c"), spaceAfter=0.3*cm, leading=16
    )
    
    # 1. 报告标题
    report_title_map = {
        "daily": "每日健康监测报告",
        "weekly": "每周健康评估报告",
        "monthly": "每月健康管理报告"
    }
    end_title_map = {
        "patient": "患者版",
        "family": "家属版",
        "doctor": "医生专业版"
    }
    title = Paragraph(f"患者{patient_id} {report_date} {report_title_map[report_type]}({end_title_map[end_type]})", title_style)
    story.append(title)
    story.append(Spacer(1, 0.5*cm))
    
    # 2. 核心总结（所有端必含）
    story.append(Paragraph("一、核心健康总结", subtitle_style))
    story.append(Paragraph(content_dict["core_summary"], normal_style))
    if content_dict.get("risk_hint"):
        story.append(Paragraph(content_dict["risk_hint"], highlight_style))
    story.append(Spacer(1, 0.3*cm))

    story.append(Paragraph("二、核心健康指标", subtitle_style))
    indicator_table_data = [["指标名称", "当前数值", "基线对比", "状态"]] + content_dict["indicator_table"]
    table = Table(indicator_table_data, colWidths=[4*cm, 3*cm, 4*cm, 3*cm])

        ("BACKGROUND", (0,1), (-1,-1), colors.HexColor("#f8f9fa")),
        ("GRID", (0,0), (-1,-1), 1, colors.HexColor("#dee2e6"))
    ]))
    story.append(table)
    story.append(Spacer(1, 0.5*cm))
    
    # 4. 可视化图片（按端适配）
    story.append(Paragraph("三、健康趋势可视化", subtitle_style))
    for img_path in content_dict[f"{end_type}_figures"]:
        if os.path.exists(img_path):
            img = Image(img_path, width=16*cm, height=10*cm)
            story.append(img)
            story.append(Spacer(1, 0.3*cm))
    
    # 5. 异常预警与建议（医生端增加详细归因，患者端简化）
    story.append(Paragraph("四、异常预警与健康建议", subtitle_style))
    if end_type == "doctor":
        story.append(Paragraph("预警归因分析：", subtitle_style))
        story.append(Paragraph(content_dict["alarm_attribution"], normal_style))
    story.append(Paragraph(content_dict["health_suggestion"], normal_style))
    
    # 生成PDF
    doc.build(story)
    print(f"{end_type}端PDF报告已生成：{doc_file}")
    return doc_file
```
